    28→/**
    29→ * Foreground service chuyên phát TTS bằng Android system TextToSpeech engine.
    30→ *
    31→ * Đặc điểm cốt lõi (đã xử lý các bug §8.5):
    32→ *  - foregroundServiceType = mediaPlayback (chạy nền khi khóa màn hình).
    33→ *  - MediaSession + Notification với 4 action: prev / play-pause / next / stop.
    34→ *  - Watchdog WATCHDOG_MS=2000: nếu speak() trả OK nhưng engine không gọi onStart
    35→ *    → retry MAX_RETRY=2 → re-init engine → onErrorInternal.
    36→ *  - Init timeout INIT_TIMEOUT_MS=6000: nếu engine không gọi onInit → tự onError.
    37→ *  - speakSeq tăng dần + currentUtteranceId guard: callback lạc/trùng bị bỏ qua.
    38→ *  - onStateChange 'playing' chỉ emit từ onStart THẬT.
    39→ *  - onResume (notification play) dùng SETTLE_MS=200 trễ sau stop().
    40→ *  - playFrom khi !ttsReady → chain qua ensureTts (không bao giờ return im lặng).
    41→ */

   127→    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
   128→        val action = intent?.action
   129→        when (action) {
   130→            ACTION_START -> handleStartAction(intent)
   131→            ACTION_PLAY_CHAPTER -> handlePlayChapterAction(intent)
   132→            ACTION_PAUSE -> onPause()
   133→            ACTION_RESUME -> onResume()
   134→            ACTION_STOP -> onStop(true)
   135→            ACTION_NEXT -> nextChapter()
   136→            ACTION_PREV -> prevChapter()
   137→            ACTION_SEEK -> {
   138→                val ch = intent?.getIntExtra(EXTRA_CHAR_INDEX, 0) ?: 0
   139→                ensureTts { playFrom(ch) }
   140→            }
   141→            ACTION_SET_RATE -> {
   142→                val r = intent?.getFloatExtra(EXTRA_RATE, 1.0f) ?: 1.0f
   143→                this.rate = r
   144→                try { tts?.setSpeechRate(r) } catch (_: Throwable) {}
   145→            }
   146→            else -> { /* no-op */ }
   147→        }
   148→        return START_STICKY
   149→    }

   163→        instance = null
   164→    }

   172→    private fun handleStartAction(intent: Intent) {
   173→        val seriesTitle = intent.getStringExtra(EXTRA_SERIES_TITLE) ?: ""
   174→        val coverUrl = intent.getStringExtra(EXTRA_COVER_URL) ?: ""
   175→        val chaptersJson = intent.getStringExtra(EXTRA_CHAPTERS_JSON) ?: "[]"
   176→        val startChapter = intent.getIntExtra(EXTRA_START_CHAPTER, 0)
   177→        val startChar = intent.getIntExtra(EXTRA_START_CHAR, 0)
   178→        val rate = intent.getFloatExtra(EXTRA_RATE, 1.0f)
   179→
   180→        this.seriesTitle = seriesTitle
   181→        this.coverUrl = coverUrl
   182→        this.chapters = parseChapters(chaptersJson)
   183→        this.chapterIndex = if (chapters.isEmpty()) 0 else startChapter.coerceIn(0, chapters.size - 1)
   184→        this.charIndex = startChar
   185→        this.announceTitle = (startChar == 0)
   186→        this.rate = rate
   187→        this.retryCount = 0
   188→
   189→        startForegroundNow()
   190→        requestAudioFocus()
   191→        updateMediaMetadata()
   192→        ensureTts { playFrom(charIndex) }
   193→    }
   194→
   195→    private fun handlePlayChapterAction(intent: Intent) {
   196→        if (chapters.isEmpty()) return
   197→        val idx = intent.getIntExtra(EXTRA_CHAPTER_INDEX, 0).coerceIn(0, chapters.size - 1)
   198→        val ch = intent.getIntExtra(EXTRA_CHAR_INDEX, 0)
   199→        this.chapterIndex = idx
   200→        this.charIndex = ch
   201→        this.announceTitle = (ch == 0)
   202→        this.retryCount = 0
   203→        updateMediaMetadata()
   204→        ensureTts { playFrom(ch) }
   205→    }

   207→    private fun parseChapters(json: String): List<ChapterInfo> {
   208→        return try {
   209→            val arr = JSONArray(json)
   210→            val out = ArrayList<ChapterInfo>(arr.length())
   211→            for (i in 0 until arr.length()) {
   212→                val o = arr.getJSONObject(i)
   213→                out.add(ChapterInfo(o.optString("title"), o.optString("content")))
   214→            }
   215→            out
   216→        } catch (_: Throwable) {
   217→            emptyList()
   218→        }
   219→    }

   225→    private fun ensureChannel() {
   226→        if (channelCreated) return
   227→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   228→            val ch = NotificationChannel(
   229→                CHANNEL_ID,
   230→                "SoNovel — Đọc truyện",
   231→                NotificationManager.IMPORTANCE_LOW
   232→            ).apply {
   233→                description = "Thông báo điều khiển nghe truyện SoNovel"
   234→                setShowBadge(false)
   235→            }
   236→            notificationManager?.createNotificationChannel(ch)
   237→        }
   238→        channelCreated = true
   239→    }

   241→    private fun startForegroundNow() {
   242→        ensureChannel()
   243→        val notif = buildNotification(playing)
   244→        try {
   245→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   246→                startForeground(NOTIF_ID, notif, ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK)
   247→            } else {
   248→                startForeground(NOTIF_ID, notif)
   249→            }
   250→        } catch (_: SecurityException) {
   251→            // Thiếu permission FOREGROUND_SERVICE_MEDIA_PLAYBACK trên Android 14+
   252→            emitError(10, "Thiếu quyền FOREGROUND_SERVICE_MEDIA_PLAYBACK")
   253→        } catch (_: Throwable) {
   254→            emitError(11, "Không thể startForeground()")
   255→        }
   256→    }

   258→    private fun buildNotification(isPlaying: Boolean): Notification {
   259→        val title = seriesTitle.ifEmpty { "SoNovel" }
   260→        val text = if (chapters.isEmpty() || chapterIndex !in chapters.indices) "Đang chuẩn bị…"
   261→                   else "Chương ${chapterIndex + 1}. ${chapters[chapterIndex].title}"
   262→
   263→        val builder = NotificationCompat.Builder(this, CHANNEL_ID)
   264→            .setSmallIcon(android.R.drawable.ic_media_play)
   265→            .setContentTitle(title)
   266→            .setContentText(text)
   267→            .setOngoing(isPlaying)
   268→            .setOnlyAlertOnce(true)
   269→            .setShowWhen(false)
   270→            .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
   271→
   272→        builder.addAction(android.R.drawable.ic_media_previous, "Trước",
   273→            buildAction(ACTION_PREV, 1))
   274→        val playPauseAction = if (isPlaying) ACTION_PAUSE else ACTION_RESUME
   275→        val playPauseIcon = if (isPlaying) android.R.drawable.ic_media_pause else android.R.drawable.ic_media_play
   276→        val playPauseLabel = if (isPlaying) "Tạm dừng" else "Phát"
   277→        builder.addAction(playPauseIcon, playPauseLabel, buildAction(playPauseAction, 2))
   278→        builder.addAction(android.R.drawable.ic_media_next, "Sau", buildAction(ACTION_NEXT, 3))
   279→        builder.addAction(android.R.drawable.ic_menu_close_clear_cancel, "Dừng",
   280→            buildAction(ACTION_STOP, 4))
   281→
   282→        try {
   283→            mediaSession?.sessionToken?.let { token ->
   284→                builder.setStyle(MediaStyle()
   285→                    .setMediaSession(token)
   286→                    .setShowActionsInCompactView(0, 1, 2))
   287→            }
   288→        } catch (_: Throwable) {}
   289→
   290→        return builder.build()
   291→    }

   293→    private fun buildAction(action: String, requestCode: Int): PendingIntent {
   294→        val intent = Intent(this, TtsService::class.java).apply { this.action = action }
   295→        val flags = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
   296→            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
   297→        } else PendingIntent.FLAG_UPDATE_CURRENT
   298→        return PendingIntent.getService(this, requestCode, intent, flags)
   299→    }

   311→    private fun setupMediaSession() {
   312→        try {
   313→            val session = MediaSessionCompat(this, "SoNovelTts")
   314→            session.setFlags(
   315→                MediaSessionCompat.FLAG_HANDLES_MEDIA_BUTTONS or
   316→                MediaSessionCompat.FLAG_HANDLES_TRANSPORT_CONTROLS
   317→            )
   318→            session.setCallback(object : MediaSessionCompat.Callback() {
   319→                override fun onPlay() { sendServiceAction(ACTION_RESUME) }
   320→                override fun onPause() { sendServiceAction(ACTION_PAUSE) }
   321→                override fun onSkipToNext() { sendServiceAction(ACTION_NEXT) }
   322→                override fun onSkipToPrevious() { sendServiceAction(ACTION_PREV) }
   323→                override fun onStop() { sendServiceAction(ACTION_STOP) }
   324→            })
   325→            session.isActive = true
   326→            mediaSession = session
   327→        } catch (_: Throwable) {}
   328→    }

   330→    private fun updateMediaMetadata() {
   331→        try {
   332→            val title = if (chapters.isEmpty() || chapterIndex !in chapters.indices) seriesTitle
   333→                else "Chương ${chapterIndex + 1}. ${chapters[chapterIndex].title}"
   334→            val md = MediaMetadataCompat.Builder()
   335→                .putString(MediaMetadataCompat.METADATA_KEY_TITLE, title)
   336→                .putString(MediaMetadataCompat.METADATA_KEY_ARTIST, seriesTitle)
   337→                .putString(MediaMetadataCompat.METADATA_KEY_ALBUM, "SoNovel")
   338→                .build()
   339→            mediaSession?.setMetadata(md)
   340→            mediaSession?.setPlaybackState(buildPlaybackState())
   341→        } catch (_: Throwable) {}
   342→    }

   344→    private fun buildPlaybackState(): PlaybackStateCompat {
   345→        val state = if (playing) PlaybackStateCompat.STATE_PLAYING else PlaybackStateCompat.STATE_PAUSED
   346→        return PlaybackStateCompat.Builder()
   347→            .setActions(
   348→                PlaybackStateCompat.ACTION_PLAY or
   349→                PlaybackStateCompat.ACTION_PAUSE or
   350→                PlaybackStateCompat.ACTION_PLAY_PAUSE or
   351→                PlaybackStateCompat.ACTION_SKIP_TO_NEXT or
   352→                PlaybackStateCompat.ACTION_SKIP_TO_PREVIOUS or
   353→                PlaybackStateCompat.ACTION_STOP
   354→            )
   355→            .setState(state, PlaybackStateCompat.PLAYBACK_POSITION_UNKNOWN, rate)

   370→    private fun requestAudioFocus() {
   371→        try {
   372→            if (afChangeListener == null) {
   373→                afChangeListener = AudioManager.OnAudioFocusChangeListener { change ->
   374→                    when (change) {
   375→                        AudioManager.AUDIOFOCUS_LOSS -> { stopSelf() }
   376→                        AudioManager.AUDIOFOCUS_LOSS_TRANSIENT,
   377→                        AudioManager.AUDIOFOCUS_LOSS_TRANSIENT_CAN_DUCK -> onPause()
   378→                        AudioManager.AUDIOFOCUS_GAIN -> onResume()
   379→                    }
   380→                }
   381→            }
   382→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   383→                val attrs = AudioAttributes.Builder()
   384→                    .setUsage(AudioAttributes.USAGE_MEDIA)
   385→                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
   386→                    .build()
   387→                val req = AudioFocusRequest.Builder(AudioManager.AUDIOFOCUS_GAIN)
   388→                    .setAudioAttributes(attrs)
   389→                    .setAcceptsDelayedFocusGain(true)
   390→                    .setOnAudioFocusChangeListener(afChangeListener!!)
   391→                    .build()
   392→                audioFocusRequest = req
   393→                audioManager?.requestAudioFocus(req)
   394→            } else {
   395→                @Suppress("DEPRECATION")
   396→                audioManager?.requestAudioFocus(
   397→                    afChangeListener,
   398→                    AudioManager.STREAM_MUSIC,
   399→                    AudioManager.AUDIOFOCUS_GAIN
   400→                )
   401→            }
   402→        } catch (_: Throwable) {}
   403→    }

   405→    private fun releaseAudioFocus() {

   421→    private fun initTts() {
   422→        try { tts?.shutdown() } catch (_: Throwable) {}
   423→        tts = null
   424→        ttsReady = false
   425→        tts = TextToSpeech(this, this)
   426→        cancelInitTimeout()
   427→        initTimeoutRunnable = Runnable {
   428→            if (!ttsReady) {
   429→                emitError(0, "TTS engine không khởi động được (init timeout)")
   430→                try { tts?.shutdown() } catch (_: Throwable) {}
   431→                tts = null
   432→                pendingPlay = null
   433→            }
   434→        }
   435→        main.postDelayed(initTimeoutRunnable!!, INIT_TIMEOUT_MS)
   436→    }
   437→
   438→    override fun onInit(status: Int) {
   439→        cancelInitTimeout()
   440→        if (status != TextToSpeech.SUCCESS) {
   441→            ttsReady = false
   442→            try { tts?.shutdown() } catch (_: Throwable) {}
   443→            tts = null
   444→            pendingPlay = null
   445→            emitError(0, "TTS engine báo lỗi khởi tạo")
   446→            return
   447→        }
   448→        val engine = tts ?: run {
   449→            ttsReady = false
   450→            return
   451→        }
   452→        ttsReady = true
   453→        engine.setAudioAttributes(
   454→            AudioAttributes.Builder()
   455→                .setUsage(AudioAttributes.USAGE_MEDIA)
   456→                .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
   457→                .build()
   458→        )
   459→        val res = engine.setLanguage(Locale("vi", "VN"))
   460→        if (res == TextToSpeech.LANG_MISSING_DATA || res == TextToSpeech.LANG_NOT_SUPPORTED) {
   461→            try { engine.setLanguage(Locale.getDefault()) } catch (_: Throwable) {}
   462→            emitError(1, "Thiếu giọng tiếng Việt")
   463→        }
   464→        try { engine.setSpeechRate(rate) } catch (_: Throwable) {}
   465→        engine.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
   466→            override fun onStart(utteranceId: String?) { handleOnStart(utteranceId) }
   467→            override fun onRangeStart(utteranceId: String?, start: Int, end: Int, frame: Int) {
   468→                handleOnRangeStart(utteranceId, start, end)
   469→            }
   470→            override fun onDone(utteranceId: String?) { handleOnDone(utteranceId) }
   471→            @Deprecated("Deprecated in Java")
   472→            override fun onError(utteranceId: String?) { handleOnError(utteranceId, -1) }
   473→            override fun onError(utteranceId: String?, errorCode: Int) {
   474→                handleOnError(utteranceId, errorCode)
   475→            }
   476→        })
   477→        pendingPlay?.invoke()
   478→        pendingPlay = null
   479→    }
   480→
   481→    private fun ensureTts(then: () -> Unit) {
   482→        if (ttsReady && tts != null) {
   483→            then()
   484→            return
   485→        }
   486→        // Chưa ready — lưu pendingPlay, nếu tts==null thì init mới
   487→        pendingPlay = then
   488→        if (tts == null) initTts()
   489→    }
   490→
   491→    private fun cancelInitTimeout() {
   492→        initTimeoutRunnable?.let { main.removeCallbacks(it) }
   493→        initTimeoutRunnable = null
   494→    }

   500→    private fun playFrom(targetChar: Int) {
   501→        if (!ttsReady || tts == null) {
   502→            // KHÔNG return im lặng — chain qua ensureTts để watchdog không mất dấu.
   503→            ensureTts { playFrom(targetChar) }
   504→            return
   505→        }
   506→        cancelWatchdog()
   507→        val engine = tts ?: return
   508→        if (chapters.isEmpty()) {
   509→            emitError(2, "Không có chương để phát")
   510→            return
   511→        }
   512→        if (chapterIndex !in chapters.indices) chapterIndex = chapters.size - 1
   513→        val content = chapters[chapterIndex].content
   514→        val clamped = targetChar.coerceIn(0, content.length)
   515→        pendingTargetChar = clamped
   516→        engineStarted = false
   517→
   518→        chunks = TtsChunker.chunk(content)
   519→        if (chunks.isEmpty()) {
   520→            emit(Events.ON_CHAPTER_END, mapOf("chapterIndex" to chapterIndex))
   521→            moveToNextChapter()
   522→            return
   523→        }
   524→        val idx = TtsChunker.findChunkIndex(chunks, clamped)
   525→        chunkIndex = if (idx >= 0) idx else 0
   526→
   527→        // Đọc tiêu đề "Chương N. Title" nếu bắt đầu chương
   528→        if (announceTitle && clamped == 0) {
   529→            val title = "Chương ${chapterIndex + 1}. ${chapters[chapterIndex].title}"
   530→            val titleId = "sonovel_title_${chapterIndex}_${++speakSeq}"
   531→            val titleParams = Bundle().apply {
   532→                putString(TextToSpeech.Engine.KEY_PARAM_UTTERANCE_ID, titleId)
   533→            }
   534→            try {
   535→                engine.speak(title, TextToSpeech.QUEUE_FLUSH, titleParams, titleId)
   536→            } catch (t: Throwable) {
   537→                emitError(3, "Không gọi được speak() tiêu đề: ${t.message}")
   538→                return
   539→            }
   540→            // Sau tiêu đề, queue chunk đầu với QUEUE_ADD
   541→            val chunkId = "sonovel_${chapterIndex}_${chunkIndex}_${++speakSeq}"
   542→            val chunkParams = Bundle().apply {
   543→                putString(TextToSpeech.Engine.KEY_PARAM_UTTERANCE_ID, chunkId)
   544→            }
   545→            try {
   546→                engine.speak(chunks[chunkIndex], TextToSpeech.QUEUE_ADD, chunkParams, chunkId)
   547→            } catch (t: Throwable) {
   548→                emitError(3, "Không gọi được speak() chunk đầu: ${t.message}")
   549→                return
   550→            }
   551→            currentUtteranceId = chunkId
   552→            armWatchdog(chunkId)
   553→            return
   554→        }
   555→
   556→        val chunkId = "sonovel_${chapterIndex}_${chunkIndex}_${++speakSeq}"
   557→        currentUtteranceId = chunkId
   558→        val params = Bundle().apply {
   559→            putString(TextToSpeech.Engine.KEY_PARAM_UTTERANCE_ID, chunkId)
   560→        }
   561→        try {
   562→            val ok = engine.speak(chunks[chunkIndex], TextToSpeech.QUEUE_FLUSH, params, chunkId)
   563→            if (ok != TextToSpeech.SUCCESS) {
   564→                handleEngineSpeakFailure()
   565→                return
   566→            }
   567→        } catch (t: Throwable) {
   568→            emitError(3, "Không gọi được speak(): ${t.message}")
   569→            return
   570→        }
   571→        armWatchdog(chunkId)
   572→    }
   573→
   574→    private fun speakNextChunk() {
   575→        val engine = tts ?: return
   576→        if (chunkIndex !in chunks.indices) return
   577→        val chunkId = "sonovel_${chapterIndex}_${chunkIndex}_${++speakSeq}"
   578→        currentUtteranceId = chunkId
   579→        engineStarted = false
   580→        val params = Bundle().apply {
   581→            putString(TextToSpeech.Engine.KEY_PARAM_UTTERANCE_ID, chunkId)
   582→        }
   583→        try {
   584→            val ok = engine.speak(chunks[chunkIndex], TextToSpeech.QUEUE_FLUSH, params, chunkId)
   585→            if (ok != TextToSpeech.SUCCESS) {
   586→                handleEngineSpeakFailure()
   587→                return
   588→            }
   589→        } catch (t: Throwable) {
   590→            emitError(3, "speak(): ${t.message}")
   591→            return
   592→        }
   593→        armWatchdog(chunkId)
   594→    }
   595→
   596→    private fun armWatchdog(expectedId: String) {
   597→        cancelWatchdog()
   598→        watchdogRunnable = Runnable {
   599→            if (currentUtteranceId == expectedId && !engineStarted) {
   600→                if (retryCount < MAX_RETRY) {
   601→                    retryCount++
   602→                    ensureTts { playFrom(pendingTargetChar) }
   603→                } else {
   604→                    retryCount = 0
   605→                    try { tts?.shutdown() } catch (_: Throwable) {}
   606→                    tts = null
   607→                    ttsReady = false
   608→                    // Re-init engine; init-timeout sẽ báo lỗi nếu vẫn treo
   609→                    ensureTts { playFrom(pendingTargetChar) }
   610→                    // Safety: nếu re-init vẫn không onStart sau 2*WATCHDOG_MS → onError
   611→                    main.postDelayed({
   612→                        if (currentUtteranceId == expectedId && !engineStarted) {
   613→                            onErrorInternal("Engine TTS không phản hồi sau re-init")
   614→                        }
   615→                    }, WATCHDOG_MS * 2)
   616→                }
   617→            }
   618→        }
   619→        main.postDelayed(watchdogRunnable!!, WATCHDOG_MS)
   620→    }
   621→
   622→    private fun cancelWatchdog() {
   623→        watchdogRunnable?.let { main.removeCallbacks(it) }
   624→        watchdogRunnable = null
   625→    }
   626→
   627→    private fun handleEngineSpeakFailure() {
   628→        if (retryCount < MAX_RETRY) {
   629→            retryCount++
   630→            main.postDelayed({ ensureTts { playFrom(pendingTargetChar) } }, SETTLE_MS)
   631→        } else {
   632→            onErrorInternal("TTS engine trả lỗi speak() quá nhiều lần")
   633→        }
   634→    }
   635→
   636→    private fun onErrorInternal(message: String) {
   637→        playing = false
   638→        cancelWatchdog()
   639→        updateNotification()
   640→        updateMediaMetadata()
   641→        emitError(99, message)
   642→    }

   644→    // -------------------------------------------------------------------
   645→    // Utterance callbacks (đã guard)
   646→    // -------------------------------------------------------------------
   647→
   648→    private fun handleOnStart(utteranceId: String?) {
   649→        if (utteranceId == null) return
   650→        if (utteranceId != currentUtteranceId) return
   651→        if (utteranceId.startsWith("sonovel_title_")) {
   652→            // Tiêu đề bắt đầu phát — không emit state change
   653→            return
   654→        }
   655→        retryCount = 0
   656→        engineStarted = true
   657→        playing = true
   658→        cancelWatchdog()
   659→        emit(Events.ON_STATE_CHANGE, mapOf("state" to "playing"))
   660→        updateNotification()
   661→        updateMediaMetadata()
   662→    }
   663→
   664→    private fun handleOnRangeStart(utteranceId: String?, start: Int, end: Int) {
   665→        if (utteranceId == null) return
   666→        if (utteranceId != currentUtteranceId) return
   667→        if (utteranceId.startsWith("sonovel_title_")) return
   668→        val offset = TtsChunker.chunkOffset(chunks, chunkIndex)
   669→        val ci = offset + start
   670→        charIndex = ci
   671→        val contentLen = if (chapterIndex in chapters.indices) chapters[chapterIndex].content.length else 0
   672→        val frac = if (contentLen > 0) ci.toFloat() / contentLen else 0f
   673→        emit(Events.ON_PROGRESS, mapOf(
   674→            "chapterIndex" to chapterIndex,
   675→            "charIndex" to ci,
   676→            "charLength" to contentLen,
   677→            "fraction" to frac
   678→        ))
   679→    }
   680→
   681→    private fun handleOnDone(utteranceId: String?) {
   682→        if (utteranceId == null) return
   683→        if (utteranceId != currentUtteranceId) return
   684→        if (utteranceId.startsWith("sonovel_title_")) {
   685→            // Tiêu đề đọc xong — chờ chunk đầu phát (đã queue qua QUEUE_ADD)
   686→            return
   687→        }
   688→        emit(Events.ON_CHUNK_DONE, mapOf(
   689→            "chapterIndex" to chapterIndex,
   690→            "chunkIndex" to chunkIndex
   691→        ))
   692→        val offset = TtsChunker.chunkOffset(chunks, chunkIndex)
   693→        charIndex = offset + (if (chunkIndex in chunks.indices) chunks[chunkIndex].length else 0)
   694→        chunkIndex++
   695→        if (chunkIndex < chunks.size) {
   696→            speakNextChunk()
   697→        } else {
   698→            announceTitle = true
   699→            charIndex = 0
   700→            emit(Events.ON_CHAPTER_END, mapOf("chapterIndex" to chapterIndex))
   701→            moveToNextChapter()
   702→        }
   703→    }
   704→
   705→    private fun handleOnError(utteranceId: String?, errorCode: Int) {
   706→        if (utteranceId == null) return
   707→        if (utteranceId != currentUtteranceId) return
   708→        if (utteranceId.startsWith("sonovel_title_")) return
   709→        emitError(errorCode, "TTS engine lỗi utterance (code=$errorCode)")
   710→    }

   712→    // -------------------------------------------------------------------
   713→    // Chapter navigation
   714→    // -------------------------------------------------------------------
   715→
   716→    private fun moveToNextChapter() {
   717→        if (chapterIndex + 1 >= chapters.size) {
   718→            playing = false
   719→            updateNotification()
   720→            updateMediaMetadata()
   721→            emit(Events.ON_SERIES_END, emptyMap<String, Any?>())
   722→            stopSelf()
   723→            return
   724→        }
   725→        chapterIndex++
   726→        chunkIndex = 0
   727→        charIndex = 0
   728→        announceTitle = true
   729→        emit(Events.ON_CHAPTER_CHANGE, mapOf("chapterIndex" to chapterIndex))
   730→        updateMediaMetadata()
   731→        playFrom(0)
   732→    }
   733→
   734→    fun nextChapter() {
   735→        if (chapters.isEmpty()) return
   736→        if (chapterIndex + 1 >= chapters.size) {
   737→            playing = false
   738→            updateNotification()
   739→            updateMediaMetadata()
   740→            emit(Events.ON_SERIES_END, emptyMap<String, Any?>())
   741→            stopSelf()
   742→            return
   743→        }
   744→        chapterIndex++
   745→        chunkIndex = 0
   746→        charIndex = 0
   747→        announceTitle = true
   748→        emit(Events.ON_CHAPTER_CHANGE, mapOf("chapterIndex" to chapterIndex))
   749→        updateMediaMetadata()
   750→        ensureTts { playFrom(0) }
   751→    }
   752→
   753→    fun prevChapter() {
   754→        if (chapters.isEmpty()) return
   755→        if (chapterIndex <= 0) return
   756→        chapterIndex--
   757→        chunkIndex = 0
   758→        charIndex = 0
   759→        announceTitle = true
   760→        emit(Events.ON_CHAPTER_CHANGE, mapOf("chapterIndex" to chapterIndex))
   761→        updateMediaMetadata()
   762→        ensureTts { playFrom(0) }
   763→    }

   766→    // Pause / Resume / Stop
   767→    // -------------------------------------------------------------------
   768→
   769→    fun onPause() {
   770→        playing = false
   771→        engineStarted = false
   772→        cancelWatchdog()
   773→        try { tts?.stop() } catch (_: Throwable) {}
   774→        emit(Events.ON_STATE_CHANGE, mapOf("state" to "paused"))
   775→        updateNotification()
   776→        updateMediaMetadata()
   777→    }
   778→
   779→    fun onResume() {
   780→        // SETTLE_MS=200 trễ sau stop() — Android TTS hay "nuốt" speak() ngay sau stop().
   781→        settleRunnable?.let { main.removeCallbacks(it) }
   782→        settleRunnable = Runnable {
   783→            if (!playing) {
   784→                ensureTts { playFrom(charIndex) }
   785→            }
   786→        }
   787→        main.postDelayed(settleRunnable!!, SETTLE_MS)
   788→    }
   789→
   790→    fun onStop(stopService: Boolean) {
   791→        playing = false
   792→        engineStarted = false
   793→        cancelWatchdog()
   794→        try { tts?.stop() } catch (_: Throwable) {}
   795→        emit(Events.ON_STATE_CHANGE, mapOf("state" to "stopped"))
   796→        updateMediaMetadata()
   797→        releaseAudioFocus()
   798→        if (stopService) {
   799→            try {
   800→                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
   801→                    stopForeground(STOP_FOREGROUND_REMOVE)
   802→                } else {
   803→                    @Suppress("DEPRECATION")
   804→                    stopForeground(true)
   805→                }
   806→            } catch (_: Throwable) {}
   807→            stopSelf()
   808→        }
   809→    }
